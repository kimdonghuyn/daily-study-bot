# 상속과 다형성을 활용한 배달 앱 상태 관리

## 모범 답안

이 문제는 **상속으로 공통 로직을 통합**하고 **다형성으로 각 배송 주체의 개별 동작을 처리**하는 전형적인 사례입니다.

### 1단계: 공통 추상화 (상속의 기초)

```python
from abc import ABC, abstractmethod
from enum import Enum
from typing import List

# 상태 변화 감지 및 알림 전송이 공통 로직
class DeliveryEntity(ABC):
    """모든 배송 주체의 부모 클래스"""
    
    def __init__(self, entity_id: str):
        self.entity_id = entity_id
        self._state = None
        self._observers: List['NotificationObserver'] = []
    
    # 🔑 공통 로직: 상태 변화 시 알림 전송
    def set_state(self, new_state: str):
        """Template Method Pattern - 상태 변화의 흐름은 동일"""
        if self._is_valid_transition(self._state, new_state):
            self._state = new_state
            self._notify_observers()  # 공통 로직
        else:
            raise ValueError(f"Invalid transition: {self._state} -> {new_state}")
    
    # 🔑 다형성 포인트: 각 주체별로 구현
    @abstractmethod
    def _is_valid_transition(self, current: str, next: str) -> bool:
        """상속받은 클래스에서 자신의 상태 규칙 정의"""
        pass
    
    def add_observer(self, observer: 'NotificationObserver'):
        self._observers.append(observer)
    
    def _notify_observers(self):
        """공통 알림 로직 - 모든 배송 주체가 동일하게 사용"""
        for observer in self._observers:
            observer.update(self.entity_id, self._state)
```

### 2단계: 각 배송 주체별 구현 (다형성)

```python
class RestaurantOrder(DeliveryEntity):
    """음식점의 주문 상태 관리"""
    VALID_TRANSITIONS = {
        None: ['preparing'],
        'preparing': ['completed'],
        'completed': ['pickup_done']
    }
    
    def _is_valid_transition(self, current: str, next: str) -> bool:
        return next in self.VALID_TRANSITIONS.get(current, [])
    
    def get_order_info(self) -> dict:
        """음식점 특화 로직"""
        return {
            'type': 'restaurant',
            'state': self._state,
            'eta_minutes': 10  # 음식점만 필요한 정보
        }


class DeliveryRider(DeliveryEntity):
    """배달원의 배달 상태 관리"""
    VALID_TRANSITIONS = {
        None: ['waiting_pickup'],
        'waiting_pickup': ['in_delivery'],
        'in_delivery': ['completed']
    }
    
    def __init__(self, rider_id: str, bike_type: str):
        super().__init__(rider_id)
        self.bike_type = bike_type
    
    def _is_valid_transition(self, current: str, next: str) -> bool:
        return next in self.VALID_TRANSITIONS.get(current, [])
    
    def get_delivery_info(self) -> dict:
        """배달원 특화 로직"""
        return {
            'type': 'rider',
            'state': self._state,
            'bike_type': self.bike_type,
            'current_location': self._get_gps()  # 배달원만 필요
        }
    
    def _get_gps(self) -> tuple:
        # GPS 조회 로직
        return (37.7749, -122.4194)


class ConvenienceStore(DeliveryEntity):
    """편의점 픽업 상태 관리"""
    VALID_TRANSITIONS = {
        None: ['preparing'],
        'preparing': ['ready'],
        'ready': ['picked_up']
    }
    
    def _is_valid_transition(self, current: str, next: str) -> bool:
        return next in self.VALID_TRANSITIONS.get(current, [])
    
    def get_store_info(self) -> dict:
        return {
            'type': 'convenience_store',
            'state': self._state,
            'available_items': 500  # 편의점만 필요
        }
```

### 3단계: 다형성을 활용한 통일된 처리

```python
class NotificationObserver(ABC):
    @abstractmethod
    def update(self, entity_id: str, state: str):
        pass


class CustomerNotificationService(NotificationObserver):
    """고객에게 알림을 보내는 서비스"""
    
    def update(self, entity_id: str, state: str):
        """모든 배송 주체의 상태 변화를 동일하게 처리"""
        message = self._translate_state(state)
        self.send_push_notification(entity_id, message)
    
    def _translate_state(self, state: str) -> str:
        translations = {
            'preparing': '준비 중입니다',
            'completed': '조리 완료!',
            'pickup_done': '픽업 완료',
            'waiting_pickup': '픽업 대기 중',
            'in_delivery': '배달 중',
        }
        return translations.get(state, '상태 변화')
    
    def send_push_notification(self, user_id: str, message: str):
        print(f"📱 [{user_id}] {message}")


# 🔑 핵심: 배송 주체의 구체적인 타입을 모르고도 처리 가능!
def process_delivery_entities(entities: List[DeliveryEntity]):
    """다형성 활용 - 모든 배송 주체를 동일하게 처리"""
    for entity in entities:
        # RestaurantOrder인지, DeliveryRider인지 알 필요 없음
        entity.set_state('some_next_state')
        # 각각의 validate 로직이 자동으로 실행됨
```

### 4단계: 실제 사용

```python
# 알림 서비스 준비
notification_service = CustomerNotificationService()

# 다양한 배송 주체들 생성
restaurant = RestaurantOrder('order_123')
rider = DeliveryRider('rider_456', 'motorcycle')
store = ConvenienceStore('store_789')

# 모든 배송 주체에 동일한 알림 서비스 등록
for entity in [restaurant, rider, store]:
    entity.add_observer(notification_service)

# 상태 변화
restaurant.set_state('preparing')   # 음식점의 규칙으로 검증
rider.set_state('waiting_pickup')   # 배달원의 규칙으로 검증
store.set_state('preparing')        # 편의점의 규칙으로 검증

# 출력:
# 📱 [order_123] 준비 중입니다
# 📱 [rider_456] 픽업 대기 중
# 📱 [store_789] 준비 중입니다
```

## 핵심 개념 정리

### 📌 상속 (Inheritance) - "코드 재사용"
- **공통 로직을 부모 클래스에 통합**
  - 상태 변화 감지, 옵저버 패턴, 알림 전송 등
  - 모든 배송 주체가 중복으로 작성할 필요 없음

- **부모 클래스의 책임**
  ```python
  class DeliveryEntity(ABC):
      - set_state()        # 공통: 상태 변화 흐름 제어
      - add_observer()     # 공통: 알림 등록
      - _notify_observers()# 공통: 알림 발송
      - _is_valid_transition() # 추상: 자식이 구현
  ```

### 📌 다형성 (Polymorphism) - "유연한 확장"
- **메서드 오버라이딩**: 각 배송 주체의 상태 규칙을 자신의 로직으로 구현
  ```
  RestaurantOrder._is_valid_transition()
  DeliveryRider._is_valid_transition()
  ConvenienceStore._is_valid_transition()
  → 같은 메서드명, 다른 구현
  ```

- **Liskov Substitution Principle**: 부모 타입으로 모든 자식을 처리 가능
  ```python
  entities: List[DeliveryEntity]  # 구체 타입 몰라도 OK
  for entity in entities:
      entity.set_state(...)  # 다형성: 각자의 validate 로직 실행
  ```
