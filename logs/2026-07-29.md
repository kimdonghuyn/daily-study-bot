```python
# 1. 추상 기본 클래스(Base Class) - 공통 기능 정의
from abc import ABC, abstractmethod
from datetime import datetime
from typing import List, Dict

class Store(ABC):
    """모든 배달 대상 가게의 기본 템플릿"""
    
    def __init__(self, store_id: str, name: str):
        self.store_id = store_id
        self.name = name
        self.orders: List[Order] = []
        self.reviews: List[Review] = []
    
    # 공통 기능 (구현된 메서드)
    def receive_order(self, order: 'Order') -> bool:
        """공통 주문 수신 로직"""
        if not self.is_open():
            raise Exception(f"{self.name}은(는) 현재 닫혀있습니다")
        
        self.orders.append(order)
        print(f"[{self.name}] 주문 #{order.order_id} 접수됨")
        return True
    
    def receive_review(self, review: 'Review') -> bool:
        """공통 리뷰 수신"""
        self.reviews.append(review)
        self._update_rating()
        return True
    
    # 추상 메서드 (각 가게마다 다르게 구현)
    @abstractmethod
    def deliver(self, order: 'Order') -> Dict:
        """배달 방식이 다름 - 반드시 각 클래스에서 구현"""
        pass
    
    @abstractmethod
    def is_open(self) -> bool:
        """영업 상태 판단 로직이 다름"""
        pass
    
    @abstractmethod
    def calculate_delivery_fee(self, distance: float) -> int:
        """배달료 계산이 다름"""
        pass
    
    # 공통 헬퍼 메서드
    def _update_rating(self):
        """평점 계산 (모든 가게가 동일)"""
        if self.reviews:
            avg = sum(r.rating for r in self.reviews) / len(self.reviews)
            self.rating = round(avg, 1)
```

### 2. 각 가게 구현 - 상속과 다형성 활용

```python
class Restaurant(Store):
    """음식점 - 일반적인 배달"""
    
    def is_open(self) -> bool:
        hour = datetime.now().hour
        return 10 <= hour < 23  # 10시~23시 영업
    
    def calculate_delivery_fee(self, distance: float) -> int:
        # 거리 기반 요금: 3km당 2,500원
        return max(2500, int(distance / 3 * 2500))
    
    def deliver(self, order: 'Order') -> Dict:
        fee = self.calculate_delivery_fee(order.distance)
        return {
            "store": self.name,
            "type": "restaurant",
            "delivery_fee": fee,
            "estimated_time": "30-40분",
            "note": "뜨거운 음식이니 빨리 드세요"
        }


class ConvenienceStore(Store):
    """편의점 - 야간 배달료 추가"""
    
    NIGHT_SURCHARGE = 3000  # 야간 할증료
    
    def is_open(self) -> bool:
        return True  # 24시간 영업
    
    def calculate_delivery_fee(self, distance: float) -> int:
        base_fee = max(1500, int(distance / 2 * 1500))
        
        # 야간(22시~06시) 할증
        hour = datetime.now().hour
        if hour >= 22 or hour < 6:
            base_fee += self.NIGHT_SURCHARGE
        
        return base_fee
    
    def deliver(self, order: 'Order') -> Dict:
        fee = self.calculate_delivery_fee(order.distance)
        hour = datetime.now().hour
        is_night = hour >= 22 or hour < 6
        
        return {
            "store": self.name,
            "type": "convenience_store",
            "delivery_fee": fee,
            "estimated_time": "20-30분",
            "night_surcharge": self.NIGHT_SURCHARGE if is_night else 0,
            "note": "상온 제품을 빠르게 배송합니다"
        }


class Cafe(Store):
    """카페 - 픽업 제약"""
    
    PICKUP_TIME_LIMIT = 5  # 분 단위
    
    def is_open(self) -> bool:
        hour = datetime.now().hour
        return 7 <= hour < 21  # 7시~21시 영업
    
    def calculate_delivery_fee(self, distance: float) -> int:
        # 카페는 가까운 거리만: 최대 2km
        if distance > 2:
            raise Exception("2km 이상은 배달 불가능합니다")
        return 1500  # 고정 요금
    
    def deliver(self, order: 'Order') -> Dict:
        if not self._check_pickup_time(order.created_at):
            raise Exception(f"{self.PICKUP_TIME_LIMIT}분 이내 픽업하지 않으면 취소됩니다")
        
        fee = self.calculate_delivery_fee(order.distance)
        return {
            "store": self.name,
            "type": "cafe",
            "delivery_fee": fee,
            "estimated_time": "10-15분",
            "pickup_deadline": f"{self.PICKUP_TIME_LIMIT}분 이내",
            "note": "신선한 음료를 빠르게 배송해드립니다"
        }
    
    def _check_pickup_time(self, created_at: datetime) -> bool:
        """픽업 시간 초과 체크"""
        elapsed = (datetime.now() - created_at).total_seconds() / 60
        return elapsed <= self.PICKUP_TIME_LIMIT
```

### 3. 실제 사용 - 다형성의 강력함

```python
class OrderService:
    """주문 처리 서비스 - 가게 종류 상관없이 통일된 인터페이스"""
    
    def process_order(self, store: Store, order: Order) -> Dict:
        """어떤 가게든 동일한 방식으로 처리"""
        try:
            # 공통 기능 사용 (상속)
            store.receive_order(order)
            
            # 각 가게별로 다른 동작 (다형성)
            delivery_info = store.deliver(order)
            
            return {
                "status": "success",
                "delivery_info": delivery_info,
                "order_id": order.order_id
            }
        except Exception as e:
            return {"status": "error", "message": str(e)}


# 실행 예시
if __name__ == "__main__":
    # 가게 생성
    stores = [
        Restaurant("r1", "맛있는 피자집"),
        ConvenienceStore("c1", "GS25 강남점"),
        Cafe("k1", "스타벅스 강남역점")
    ]
    
    service = OrderService()
    order = Order("ord123", distance=3.5)
    
    # 모든 가게를 동일한 방식으로 처리
    for store in stores:
        result = service.process_order(store, order)
        print(f"\n{store.name}:")
        print(f"배달료: {result['delivery_info']['delivery_fee']}원")
        print(f"예상시간: {result['delivery_info']['estimated_time']}")
```

- **이 경우**: `Store` 클래스에 `receive_order()`, `receive_review()` 등 공통 기능 정의
- **장점**: 한 곳에서 수정하면 모든 자식 클래스에 반영됨
