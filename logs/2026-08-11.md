from abc import ABC, abstractmethod

# 1️⃣ 기본 캐릭터 클래스 (부모 클래스)
class Character(ABC):
    def __init__(self, name: str, hp: int, level: int):
        self.name = name
        self.hp = hp
        self.max_hp = hp
        self.level = level
        self.exp = 0
    
    # 🔄 공통 메서드
    def take_damage(self, damage: int):
        self.hp = max(0, self.hp - damage)
        print(f"{self.name}이(가) {damage}의 피해를 입었습니다. 남은 HP: {self.hp}")
    
    def gain_exp(self, exp: int):
        self.exp += exp
        if self.exp >= 100:
            self.level_up()
    
    def level_up(self):
        self.level += 1
        self.exp = 0
        self.max_hp += 20
        self.hp = self.max_hp
        print(f"🎉 {self.name}이(가) 레벨 {self.level}로 상승했습니다!")
    
    def show_status(self):
        print(f"\n{'='*40}")
        print(f"이름: {self.name} | 클래스: {self.__class__.__name__}")
        print(f"HP: {self.hp}/{self.max_hp} | 레벨: {self.level}")
        print(f"경험치: {self.exp}/100")
        print(f"{'='*40}")
    
    # 🎯 다형성: 자식 클래스에서 구현해야 할 메서드
    @abstractmethod
    def attack(self, target: 'Character') -> int:
        """각 클래스마다 다르게 구현되어야 함"""
        pass
    
    @abstractmethod
    def special_ability(self, target: 'Character'):
        """특수 능력 (자식 클래스에서 구현)"""
        pass


# 2️⃣ 워리어 클래스 (자식 클래스)
class Warrior(Character):
    def __init__(self, name: str):
        super().__init__(name, hp=150, level=1)
        self.armor = 10  # 워리어만의 특성
    
    def attack(self, target: Character) -> int:
        """검 공격"""
        damage = 25 + self.level * 5
        print(f"⚔️  {self.name}이(가) {target.name}에게 검으로 공격합니다!")
        target.take_damage(damage)
        return damage
    
    def special_ability(self, target: Character):
        """방어 태세 (다음 공격 감소)"""
        damage = 40 + self.level * 8
        print(f"🛡️  {self.name}이(가) 무차별 베기를 사용합니다!")
        target.take_damage(damage)
        self.gain_exp(20)


# 3️⃣ 마법사 클래스 (자식 클래스)
class Mage(Character):
    def __init__(self, name: str):
        super().__init__(name, hp=80, level=1)
        self.mana = 100
        self.max_mana = 100
    
    def attack(self, target: Character) -> int:
        """마법 공격"""
        if self.mana < 20:
            print(f"❌ {self.name}의 마나가 부족합니다!")
            return 0
        
        damage = 35 + self.level * 7
        self.mana -= 20
        print(f"🔥 {self.name}이(가) 파이어볼을 시전합니다! (마나: {self.mana})")
        target.take_damage(damage)
        return damage
    
    def special_ability(self, target: Character):
        """텔레포트 (공격 회피 후 반격)"""
        if self.mana < 50:
            print(f"❌ 마나가 부족합니다!")
            return
        
        self.mana -= 50
        damage = 60 + self.level * 10
        print(f"✨ {self.name}이(가) 텔레포트로 회피하고 반격합니다!")
        target.take_damage(damage)
        self.gain_exp(30)
    
    def restore_mana(self):
        """마나 회복"""
        self.mana = self.max_mana
        print(f"💧 {self.name}의 마나가 회복되었습니다.")


# 4️⃣ 궁수 클래스 (자식 클래스)
class Archer(Character):
    def __init__(self, name: str):
        super().__init__(name, hp=100, level=1)
        self.arrows = 30
    
    def attack(self, target: Character) -> int:
        """화살 공격"""
        if self.arrows <= 0:
            print(f"❌ {self.name}의 화살이 부족합니다!")
            return 0
        
        damage = 20 + self.level * 4
        self.arrows -= 1
        print(f"🏹 {self.name}이(가) 화살을 쏩니다! (남은 화살: {self.arrows})")
        target.take_damage(damage)
        return damage
    
    def special_ability(self, target: Character):
        """분산 사격 (여러 번 공격)"""
        if self.arrows < 5:
            print(f"❌ 화살이 부족합니다!")
            return
        
        self.arrows -= 5
        damage = 15 * 3  # 3번 공격
        print(f"🎯 {self.name}이(가) 분산 사격을 시전합니다!")
        for i in range(3):
            target.take_damage(15)
        self.gain_exp(25)
    
    def restock_arrows(self):
        """화살 보충"""
        self.arrows = 30
        print(f"📦 {self.name}이(가) 화살을 보충했습니다.")


# 5️⃣ 새로운 캐릭터 추가하기 (확장성)
class Assassin(Character):
    """새로운 캐릭터를 쉽게 추가할 수 있음"""
    def __init__(self, name: str):
        super().__init__(name, hp=90, level=1)
        self.stealth = True
    
    def attack(self, target: Character) -> int:
        """은신 상태에서의 공격"""
        if self.stealth:
            damage = 50 + self.level * 10  # 은신 상태 보너스
            print(f"🗡️  {self.name}이(가) 은신 상태에서 습격합니다!")
            self.stealth = False
        else:
            damage = 20 + self.level * 4
            print(f"🗡️  {self.name}이(가) 공격합니다!")
        
        target.take_damage(damage)
        return damage
    
    def special_ability(self, target: Character):
        """그림자 암살"""
        damage = 80 + self.level * 12
        print(f"💀 {self.name}이(가) 그림자 암살을 시전합니다!")
        target.take_damage(damage)
        self.stealth = True
        self.gain_exp(35)
